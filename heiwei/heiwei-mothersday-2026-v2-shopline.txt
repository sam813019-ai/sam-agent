(function () {
  var root = document.querySelector('.heiwei-md2026');
  if (!root) return;

  // 1) Scroll reveal
  var items = root.querySelectorAll('[data-reveal]');
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
    items.forEach(function (el) { io.observe(el); });
  } else {
    items.forEach(function (el) { el.classList.add('is-visible'); });
  }

  // 2) 倒數 — 2026/5/10
  var target = new Date('2026-05-10T00:00:00+08:00').getTime();
  var timerEl = document.getElementById('heiweiCdTimer');
  function pad(n){ return String(n).padStart(2,'0'); }
  function tick() {
    if (!timerEl) return;
    var diff = target - Date.now();
    if (diff <= 0) { timerEl.textContent = '母親節快樂 · Happy Mother\u2019s Day'; return; }
    var d = Math.floor(diff / 86400000);
    var h = Math.floor((diff % 86400000) / 3600000);
    var m = Math.floor((diff % 3600000) / 60000);
    var s = Math.floor((diff % 60000) / 1000);
    timerEl.textContent = '距 5/10 · ' + d + ' Days ' + pad(h) + ':' + pad(m) + ':' + pad(s);
  }
  tick(); setInterval(tick, 1000);

  // 3) Floating CTA
  var fcta = document.getElementById('heiweiFcta');
  var hero = root.querySelector('.hero');
  function toggleFcta() {
    if (!fcta || !hero) return;
    var triggerY = hero.getBoundingClientRect().top + window.scrollY + hero.offsetHeight * 0.8;
    if (window.scrollY > triggerY) fcta.classList.add('is-shown');
    else fcta.classList.remove('is-shown');
  }
  toggleFcta();
  window.addEventListener('scroll', toggleFcta, { passive: true });

  // 4) Banner 輕視差
  if (window.matchMedia('(min-width: 821px) and (prefers-reduced-motion: no-preference)').matches) {
    var banners = root.querySelectorAll('.editorial-banner .eb-image img');
    window.addEventListener('scroll', function () {
      banners.forEach(function (img) {
        var rect = img.getBoundingClientRect();
        var vh = window.innerHeight;
        if (rect.bottom > 0 && rect.top < vh) {
          var progress = (vh - rect.top) / (vh + rect.height);
          var shift = (progress - 0.5) * 14;
          img.style.transform = 'scale(1.02) translateY(' + shift.toFixed(1) + 'px)';
        }
      });
    }, { passive: true });
  }
})();
