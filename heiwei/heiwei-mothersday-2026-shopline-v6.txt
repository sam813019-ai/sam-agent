  var items = root.querySelectorAll('[data-reveal]');
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
    items.forEach(function (el) { io.observe(el); });
  } else {
    items.forEach(function (el) { el.classList.add('is-visible'); });
  }

  // 2) 母親節倒數 — 台灣母親節 2026/5/10
  var target = new Date('2026-05-10T00:00:00+08:00').getTime();
  var timerEl = document.getElementById('cdTimer');
  function tick() {
    if (!timerEl) return;
    var diff = target - Date.now();
    if (diff <= 0) { timerEl.textContent = '母親節快樂 ♡'; return; }
    var d = Math.floor(diff / 86400000);
    var h = Math.floor((diff % 86400000) / 3600000);
    var m = Math.floor((diff % 3600000) / 60000);
    var s = Math.floor((diff % 60000) / 1000);
    timerEl.textContent = '距 5/10 母親節 · ' + d + '天 ' +
      String(h).padStart(2, '0') + ':' +
      String(m).padStart(2, '0') + ':' +
      String(s).padStart(2, '0');
  }
  tick();
  setInterval(tick, 1000);

  // 3) 浮動 CTA —— 滾過 hero 才出現
  var fcta = document.getElementById('floatingCta');
  var hero = root.querySelector('.hero-intro');
  function toggleFcta() {
    if (!fcta || !hero) return;
    var heroRect = hero.getBoundingClientRect();
    var heroBottomInPage = window.scrollY + heroRect.bottom;
    if (window.scrollY > heroBottomInPage - window.innerHeight * 0.5) {
      fcta.classList.add('is-shown');
    } else {
      fcta.classList.remove('is-shown');
    }
  }
  toggleFcta();
  window.addEventListener('scroll', toggleFcta, { passive: true });

  // 3.5) 點擊播放有聲影片（無控制欄、不重複播、iOS 有聲）
  // 預載 Vimeo API：確保使用者點擊時 API 已就緒，iOS 需在同一 user gesture 呼叫 play()
  if (!window._hwVimeoLoading) {
    window._hwVimeoLoading = true;
    var _hvs = document.createElement('script');
    _hvs.src = 'https://player.vimeo.com/api/player.js';
    document.head.appendChild(_hvs);
  }

  window.hwLoadVimeo = function (el) {
    var id = el.getAttribute('data-vimeo');
    var container = el.parentNode;

    var wrapper = document.createElement('div');
    wrapper.className = 'hw-vimeo-wrapper';

    var iframe = document.createElement('iframe');
    // 不用 autoplay=1；改由 player.play() 在 user gesture 內觸發，iOS Safari 才有聲音
    iframe.src = 'https://player.vimeo.com/video/' + id +
      '?badge=0&autopause=0&loop=0&controls=0&title=0&byline=0&portrait=0&playsinline=1';
    iframe.className = 'hw-vimeo-player';
    iframe.setAttribute('frameborder', '0');
    iframe.setAttribute('allow', 'autoplay; fullscreen; picture-in-picture');
    iframe.setAttribute('title', '品牌影片');
    wrapper.appendChild(iframe);

    var endOverlay = document.createElement('div');
    endOverlay.className = 'hw-end-overlay';
    wrapper.appendChild(endOverlay);

    var ctrl = document.createElement('div');
    ctrl.className = 'hw-vimeo-ctrl';
    ctrl.setAttribute('role', 'button');
    ctrl.setAttribute('aria-label', '播放 / 暫停');
    ctrl.innerHTML =
      '<svg class="hw-icon-pause hw-hidden" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 80 80">' +
        '<circle cx="40" cy="40" r="38" fill="rgba(0,0,0,0.5)" stroke="rgba(255,255,255,0.8)" stroke-width="2"/>' +
        '<rect x="27" y="24" width="9" height="32" rx="2" fill="#fff"/>' +
        '<rect x="44" y="24" width="9" height="32" rx="2" fill="#fff"/>' +
      '</svg>' +
      '<svg class="hw-icon-play" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 80 80">' +
        '<circle cx="40" cy="40" r="38" fill="rgba(0,0,0,0.5)" stroke="rgba(255,255,255,0.8)" stroke-width="2"/>' +
        '<polygon points="33,24 61,40 33,56" fill="#fff"/>' +
      '</svg>';
    wrapper.appendChild(ctrl);
    container.replaceChild(wrapper, el);

    function setupPlayer(player) {
      var isEnded = false;

      // play() 在此直接被 user gesture 的 call stack 呼叫（iOS 有聲關鍵）
      player.play().catch(function () {});

      ctrl.addEventListener('click', function () {
        endOverlay.style.display = 'none';
        if (isEnded) {
          isEnded = false;
          player.setCurrentTime(0);
          player.play().catch(function () {});
        } else {
          player.getPaused().then(function (paused) {
            if (paused) { player.play(); } else { player.pause(); }
          });
        }
      });

      player.on('play', function () {
        ctrl.querySelector('.hw-icon-pause').classList.remove('hw-hidden');
        ctrl.querySelector('.hw-icon-play').classList.add('hw-hidden');
      });
      player.on('pause', function () {
        ctrl.querySelector('.hw-icon-pause').classList.add('hw-hidden');
        ctrl.querySelector('.hw-icon-play').classList.remove('hw-hidden');
      });
      player.on('ended', function () {
        isEnded = true;
        endOverlay.style.display = 'block';   // 蓋住 Vimeo 推薦影片
        ctrl.querySelector('.hw-icon-pause').classList.add('hw-hidden');
        ctrl.querySelector('.hw-icon-play').classList.remove('hw-hidden');
      });
    }

    if (window.Vimeo && window.Vimeo.Player) {
      setupPlayer(new Vimeo.Player(iframe));  // API 已就緒：直接在 user gesture 內執行
    } else {
      // API 尚未載入（慢速網路）：polling 等待，之後 ctrl 點擊仍在 user gesture 內
      var pollT = 0;
      var poll = setInterval(function () {
        pollT += 100;
        if (window.Vimeo && window.Vimeo.Player) {
          clearInterval(poll);
          setupPlayer(new Vimeo.Player(iframe));
        } else if (pollT >= 8000) { clearInterval(poll); }
      }, 100);
    }
  };

  // 4) Banner 輕視差（僅桌機）
  if (window.matchMedia('(min-width: 769px) and (prefers-reduced-motion: no-preference)').matches) {
    var bannerItems = root.querySelectorAll('.shopline-split-banner .banner-item');
    window.addEventListener('scroll', function () {
      bannerItems.forEach(function (item) {
        var rect = item.getBoundingClientRect();
        var vh = window.innerHeight;
        if (rect.bottom > 0 && rect.top < vh) {
          var progress = (vh - rect.top) / (vh + rect.height);
          var shift = (progress - 0.5) * 12;
          item.style.transform = 'translateY(' + shift.toFixed(1) + 'px)';
        }
      });
    }, { passive: true });
  }
})();
</script>
